# là bỏ lời hứa đó.
#
#   python -m venv .venv-audio
#   .venv-audio\Scripts\activate          # Windows
#   source .venv-audio/bin/activate       # macOS / Linux
#   python -m pip install -r tools/audio/requirements.txt
